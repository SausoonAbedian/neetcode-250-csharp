    Assert.False(s.Contains(1));
    s.Remove(0);
    s.Remove(12345); // removing a missing key is a no-op
    Assert.False(s.Contains(0));
    Assert.True(s.Contains(997));
}

    {
        throw new NotImplementedException();

## Tests
[Fact]
public void Example1()
{
    var m = new MyHashMap();
    m.Put(1, 1);
    m.Put(2, 2);
    Assert.Equal(1, m.Get(1));
    Assert.Equal(-1, m.Get(3));
    m.Put(2, 1);
    Assert.Equal(1, m.Get(2));
    m.Remove(2);
    Assert.Equal(-1, m.Get(2));
}
[Fact]
public void CollisionsAndOverwrite()
{
    var m = new MyHashMap();
    for (var i = 0; i < 5000; i++) m.Put(i * 200, i);
    for (var i = 0; i < 5000; i += 2) m.Remove(i * 200);
    m.Put(200, 42);
    Assert.Equal(42, m.Get(200));
    Assert.Equal(-1, m.Get(0));
    Assert.Equal(4999, m.Get(999_800));
    Assert.Equal(-1, m.Get(1_000_000));
}
