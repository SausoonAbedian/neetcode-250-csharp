public void DotsAndLengths()
{
    var d = new WordDictionary();
    d.AddWord("a");
    d.AddWord("ab");
    Assert.True(d.Search("."));

    Assert.False(d.Search("..."));
    Assert.False(d.Search(".a"));
    Assert.True(d.Search("a."));
}
