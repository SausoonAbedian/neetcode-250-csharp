[Fact]
public void Example1()
{
    var t = new Twitter();
    t.PostTweet(1, 5);
    Assert.Equal(new[] { 5 }, t.GetNewsFeed(1));
    t.Follow(1, 2);
    t.PostTweet(2, 6);

    t.Unfollow(1, 2);
    Assert.Equal(new[] { 5 }, t.GetNewsFeed(1));
}
[Fact]
public void FeedLimitedToTenNewest()
{
    var t = new Twitter();
    for (var i = 0; i < 15; i++) t.PostTweet(i % 3 + 1, 100 + i);
    t.Follow(1, 2);
    t.Follow(1, 3);
    Assert.Equal(Enumerable.Range(105, 10).Reverse().ToArray(), t.GetNewsFeed(1));
}
[Fact]
public void EdgeCases()
{
    var t = new Twitter();
    Assert.Empty(t.GetNewsFeed(42));
    t.PostTweet(1, 1);
    t.Follow(1, 1);
    t.Unfollow(1, 1); // must not hide your own tweets
    t.Unfollow(1, 99); // unfollowing someone you don't follow is a no-op
    Assert.Equal(new[] { 1 }, t.GetNewsFeed(1));
    t.Follow(2, 1);
    t.Follow(2, 1);
    Assert.Equal(new[] { 1 }, t.GetNewsFeed(2));
}
