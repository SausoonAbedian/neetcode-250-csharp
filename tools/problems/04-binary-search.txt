/**
 * The guess API is defined in the parent class GuessGame (provided below, do not edit).
 * int guess(int num) returns -1 if num is higher than the pick,
 *                            1 if num is lower than the pick,
 *                            0 if num == pick.
 */
public class Solution : GuessGame
{
    public Solution(int pick) : base(pick) { }

    public int GuessNumber(int n)
    {
        throw new NotImplementedException();
    }
}

// ---- provided ----
public class GuessGame
{
    private readonly int _pick;
    public int Calls { get; private set; }
    public GuessGame(int pick) => _pick = pick;

    public int guess(int num)
    {
        Calls++;
        return num > _pick ? -1 : num < _pick ? 1 : 0;
    }
}
## Tests
[Fact] public void Example1() => Assert.Equal(6, new Solution(6).GuessNumber(10));
[Fact] public void Example2() => Assert.Equal(1, new Solution(1).GuessNumber(1));
[Fact] public void Example3() => Assert.Equal(1, new Solution(1).GuessNumber(2));
[Fact]
public void LargeNoOverflowAndLogarithmic()
{
    var s = new Solution(int.MaxValue - 1);
    Assert.Equal(int.MaxValue - 1, s.GuessNumber(int.MaxValue));
    Assert.True(s.Calls <= 40, $"Used {s.Calls} guesses; binary search needs ~31.");
}

## Solution
/**
 * This is MountainArray's API interface (provided below, do not edit).
 * You should not implement it, or speculate about its implementation.
 * At most 100 calls to get() are allowed.
 */
public class Solution
{
    public int FindInMountainArray(int target, MountainArray mountainArr)
    {
        throw new NotImplementedException();
    }
}

// ---- provided ----
public class MountainArray
{
    private readonly int[] _a;
    public int GetCalls { get; private set; }
    public MountainArray(params int[] a) => _a = a;

    public int Get(int index)
    {
        if (++GetCalls > 100) throw new InvalidOperationException("More than 100 calls to Get().");
        return _a[index];
    }

    public int Length() => _a.Length;
}
## Tests
[Fact] public void Example1() => Assert.Equal(2, new Solution().FindInMountainArray(3, new MountainArray(1, 2, 3, 4, 5, 3, 1)));
[Fact] public void Example2() => Assert.Equal(-1, new Solution().FindInMountainArray(3, new MountainArray(0, 1, 2, 4, 2, 1)));
[Fact] public void OnlyOnRightSide() => Assert.Equal(4, new Solution().FindInMountainArray(2, new MountainArray(1, 5, 9, 4, 2)));
[Fact] public void Peak() => Assert.Equal(2, new Solution().FindInMountainArray(9, new MountainArray(1, 5, 9, 4, 2)));
[Fact]
public void LargeArrayWithinCallLimit()
{
    var a = Enumerable.Range(0, 5000).Concat(Enumerable.Range(0, 4999).Reverse()).ToArray();
    Assert.Equal(1234, new Solution().FindInMountainArray(1234, new MountainArray(a)));
    var b = Enumerable.Range(0, 5000).Select(x => x * 2).Concat(Enumerable.Range(0, 5000).Reverse().Select(x => x * 2 + 1)).ToArray();
    Assert.Equal(5000 + (5000 - 1 - 1234), new Solution().FindInMountainArray(2469, new MountainArray(b)));
}
