// NeetCode version ("Foreign Dictionary"): words are sorted by an unknown alphabet.
// Return any order of ALL unique letters that is consistent with the words, or "" if none exists.
public class Solution
{
    public string ForeignDictionary(string[] words)
    {
        throw new NotImplementedException();
    }
}
## Tests
private static void CheckValid(params string[] words)
{
    var r = new Solution().ForeignDictionary(words);
    var letters = words.SelectMany(w => w).Distinct().OrderBy(c => c).ToArray();
    Assert.Equal(letters, r.OrderBy(c => c).ToArray());
    var pos = r.Select((c, i) => (c, i)).ToDictionary(x => x.c, x => x.i);
    for (var i = 0; i + 1 < words.Length; i++)
    {
        var a = words[i];
        var b = words[i + 1];
        for (var j = 0; j < Math.Min(a.Length, b.Length); j++)
        {
            if (a[j] == b[j]) continue;
            Assert.True(pos[a[j]] < pos[b[j]], $"'{a[j]}' must come before '{b[j]}' in \"{r}\"");
            break;
        }
    }
}
[Fact] public void Example1() => CheckValid("z", "o");
[Fact] public void Example2() => CheckValid("hrn", "hrf", "er", "enn", "rfnn");
[Fact] public void LeetCodeExample() => CheckValid("wrt", "wrf", "er", "ett", "rftt");
[Fact] public void Cycle() => Assert.Equal("", new Solution().ForeignDictionary(new[] { "z", "x", "z" }));
[Fact] public void PrefixAfterLongerWordIsInvalid() => Assert.Equal("", new Solution().ForeignDictionary(new[] { "abc", "ab" }));
[Fact] public void SameWord() => CheckValid("abc", "abc");

## Tests
private static void CheckValid(int k, string rows, string cols)
{
    var m = new Solution().BuildMatrix(k, Lc.Grid(rows), Lc.Grid(cols));
    Assert.Equal(k, m.Length);
    var r = new int[k + 1];
    var c = new int[k + 1];
    var seen = 0;
    for (var i = 0; i < k; i++)
    {
        Assert.Equal(k, m[i].Length);
        for (var j = 0; j < k; j++)
        {
            if (m[i][j] == 0) continue;
            seen++;
            r[m[i][j]] = i;
            c[m[i][j]] = j;
        }
    }
    Assert.Equal(k, seen);
    Assert.Equal(k, Enumerable.Range(0, k).SelectMany(i => m[i]).Where(v => v != 0).Distinct().Count());
    foreach (var p in Lc.Grid(rows)) Assert.True(r[p[0]] < r[p[1]], $"{p[0]} must be in a row above {p[1]}");
    foreach (var p in Lc.Grid(cols)) Assert.True(c[p[0]] < c[p[1]], $"{p[0]} must be in a column left of {p[1]}");
}
[Fact] public void Example1() => CheckValid(3, "[[1,2],[3,2]]", "[[2,1],[3,2]]");
[Fact] public void Example2() => Assert.Empty(new Solution().BuildMatrix(3, Lc.Grid("[[1,2],[2,3],[3,1],[2,3]]"), Lc.Grid("[[2,1]]")));
[Fact] public void NoConditions() => CheckValid(4, "[]", "[]");
[Fact] public void ColumnCycle() => Assert.Empty(new Solution().BuildMatrix(2, Lc.Grid("[[1,2]]"), Lc.Grid("[[1,2],[2,1]]")));
