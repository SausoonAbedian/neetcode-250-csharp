public class Node

    }

## Tests
// adjList[i] = neighbors of node (i + 1)
private static Node Build(int[][] adj)
{
    if (adj.Length == 0) return null;
    var nodes = Enumerable.Range(1, adj.Length).Select(i => new Node(i)).ToArray();
    for (var i = 0; i < adj.Length; i++)
        foreach (var n in adj[i]) nodes[i].neighbors.Add(nodes[n - 1]);
    return nodes[0];
}
private static void Check(string json)
{
    var adj = Lc.Grid(json);
    var original = Build(adj);
    var originals = new HashSet<Node>();
    void Collect(Node n) { if (n == null || !originals.Add(n)) return; foreach (var x in n.neighbors) Collect(x); }
    Collect(original);
    var clone = new Solution().CloneGraph(original);
    if (adj.Length == 0) { Assert.Null(clone); return; }
    var seen = new Dictionary<int, Node>();
    var stack = new Stack<Node>();
    stack.Push(clone);
    while (stack.Count > 0)
    {
        var n = stack.Pop();
        if (seen.ContainsKey(n.val)) { Assert.Same(seen[n.val], n); continue; }
        Assert.DoesNotContain(n, originals);
        seen[n.val] = n;
        Assert.Equal(adj[n.val - 1], n.neighbors.Select(x => x.val).ToArray());
        foreach (var x in n.neighbors) stack.Push(x);
    }
    Assert.Equal(adj.Length, seen.Count);
}
[Fact] public void Example1() => Check("[[2,4],[1,3],[2,4],[1,3]]");
[Fact] public void Example2() => Check("[[]]");
[Fact] public void Example3() => Check("[]");
[Fact] public void Line() => Check("[[2],[1,3],[2]]");
