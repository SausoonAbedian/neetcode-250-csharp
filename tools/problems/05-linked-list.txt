## Tests
// Each entry is [val, randomIndex] where randomIndex may be null.
private static Node Build(int?[][] spec)
{
    var nodes = spec.Select(s => new Node(s[0].Value)).ToArray();
    for (var i = 0; i < nodes.Length; i++)
    {
        if (i + 1 < nodes.Length) nodes[i].next = nodes[i + 1];
        if (spec[i][1] != null) nodes[i].random = nodes[spec[i][1].Value];
    }
    return nodes.FirstOrDefault();
}
private static void Check(int?[][] spec)
{
    var head = Build(spec);
    var originals = new HashSet<Node>();
    for (var n = head; n != null; n = n.next) originals.Add(n);
    var copy = new Solution().CopyRandomList(head);
    var copies = new List<Node>();
    for (var n = copy; n != null; n = n.next) copies.Add(n);
    Assert.Equal(spec.Length, copies.Count);
    for (var i = 0; i < spec.Length; i++)
    {
        Assert.DoesNotContain(copies[i], originals);
        Assert.Equal(spec[i][0], copies[i].val);
        var r = copies[i].random;
        if (spec[i][1] == null) Assert.Null(r);
        else Assert.Same(copies[spec[i][1].Value], r);
    }
    // original list must be left intact
    var orig = new List<Node>();
    for (var n = head; n != null; n = n.next) orig.Add(n);
    Assert.Equal(spec.Length, orig.Count);
    for (var i = 0; i < spec.Length; i++)
        Assert.Same(spec[i][1] == null ? null : orig[spec[i][1].Value], orig[i].random);
}
[Fact] public void Example1() => Check(new[] { new int?[] { 7, null }, new int?[] { 13, 0 }, new int?[] { 11, 4 }, new int?[] { 10, 2 }, new int?[] { 1, 0 } });
[Fact] public void Example2() => Check(new[] { new int?[] { 1, 1 }, new int?[] { 2, 1 } });
[Fact] public void Example3() => Check(new[] { new int?[] { 3, null }, new int?[] { 3, 0 }, new int?[] { 3, null } });
[Fact] public void Empty() => Assert.Null(new Solution().CopyRandomList(null));

    Assert.Equal(5, c.Get(1));
