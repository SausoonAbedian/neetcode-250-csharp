// Count axis-aligned squares with positive area formed by the query point and three stored points.
// Duplicate stored points count separately.

{
    public DetectSquares()
    {
    }

    public void Add(int[] point)
    {
        throw new NotImplementedException();
    }

    public int Count(int[] point)
    {
        throw new NotImplementedException();
    }
}
## Tests
[Fact]
public void Example1()
{
    var d = new DetectSquares();
    d.Add(new[] { 3, 10 });
    d.Add(new[] { 11, 2 });
    d.Add(new[] { 3, 2 });
    Assert.Equal(1, d.Count(new[] { 11, 10 }));
    Assert.Equal(0, d.Count(new[] { 14, 8 }));
    d.Add(new[] { 11, 2 });
    Assert.Equal(2, d.Count(new[] { 11, 10 }));
}
[Fact]
public void BothDirectionsAndNoZeroArea()
{
    var d = new DetectSquares();
    d.Add(new[] { 0, 0 });
    d.Add(new[] { 0, 1 });
    d.Add(new[] { 1, 0 });
    d.Add(new[] { 2, 0 });
    d.Add(new[] { 2, 1 });
    Assert.Equal(2, d.Count(new[] { 1, 1 })); // square to the left and to the right
    d.Add(new[] { 5, 5 });
    Assert.Equal(0, d.Count(new[] { 5, 5 })); // same point, zero area
}
